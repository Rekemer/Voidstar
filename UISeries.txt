#define UI_BackgroundColor(v) UI_DeferLoop(UI_PushBackgroundColor(v), UI_PopBackgroundColor())

#define UI_DeferLoop(begin, end) for(int _i_ = ((begin), 0); !_i_; _i_ += 1, (end))
